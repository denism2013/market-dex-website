function resampleSecs(secs, m) {
  const keys = ['z', 'w', 'hT', 'hB', 'e', 'yc'], out = [], n = secs.length, cr = (a, b, c, d, t) => .5 * ((2 * b) + (-a + c) * t + (2 * a - 5 * b + 4 * c - d) * t * t + (-a + 3 * b - 3 * c + d) * t * t * t);
  for (let k = 0; k < m; k++) { const f = k / (m - 1) * (n - 1), i = Math.min(n - 2, Math.floor(f)), t = f - i, s = {}; for (const key of keys) s[key] = cr(secs[Math.max(0, i - 1)][key], secs[i][key], secs[i + 1][key], secs[Math.min(n - 1, i + 2)][key], t); s.w = Math.max(.012, s.w); s.hT = Math.max(.012, s.hT); s.hB = Math.max(.012, s.hB); out.push(s); }
  return out;

function marketStory(w) {

  let hi = 0; for (let j = 0; j <= w; j++) hi = Math.max(hi, spy[j]);
  const f = { w, date: fmtD(R.dates[w]), ytd: k > 0 ? spy[w] / spy[k - 1] - 1 : NaN, r1: rr(4), r3: rr(13), vol: R.vol[w], dd: spy[w] / hi - 1, vis, cnt, by };
  const names = a => a.map(p => SHORT[p.meta.t]), lead = by[0].slice().sort((a, b) => b.ratio - a.ratio), defLead = lead.filter(p => DEF.has(p.meta.t)), growLead = lead.filter(p => GROWTH.has(p.meta.t));

const _vc = new V3(), _qi = new THREE.Quaternion();
function updateSpeedLines(dt) {
  const L = RIDE.lines; if (!L) return; const s = RIDE.on && (RIDE.phase === 'fly' || (RIDE.phase === 'leave' && RIDE.view === 'cockpit')) ? Math.max(RIDE.speedN, RIDE.jumpN) : 0; L.visible = s > .12; if (!L.visible) return;
  const chase = RIDE.view !== 'cockpit'; L.material.opacity = Math.min(1, (s - .12) * 1.6) * (chase ? .4 + .6 * RIDE.jumpN : 1); L.geometry.setDrawRange(0, (chase && RIDE.jumpN < .3 ? Math.floor(RIDE.lineN / 2) : RIDE.lineN) * 2);
  _qi.copy(camera.quaternion).invert(); _vc.copy(SHIP.vel).applyQuaternion(_qi); const sp = _vc.length(); if (sp < 1e-3) return; _vc.multiplyScalar(1 / sp);
  const pos = L.geometry.attributes.position.array, sd = RIDE.lineSeed, move = sp * dt * 1.4, len = .5 + s * 6 + RIDE.jumpN * 18;
  for (let i = 0; i < RIDE.lineN; i++) {
    let x = sd[i * 3] - _vc.x * move, y = sd[i * 3 + 1] - _vc.y * move, z = sd[i * 3 + 2] - _vc.z * move;
    if (z > -1) z -= 59; else if (z < -60) z += 59; if (x > 8) x -= 16; else if (x < -8) x += 16; if (y > 5) y -= 10; else if (y < -5) y += 10;
    sd[i * 3] = x; sd[i * 3 + 1] = y; sd[i * 3 + 2] = z;
    pos[i * 6] = x; pos[i * 6 + 1] = y; pos[i * 6 + 2] = z; pos[i * 6 + 3] = x + _vc.x * len; pos[i * 6 + 4] = y + _vc.y * len; pos[i * 6 + 5] = z + _vc.z * len;
  }
  L.geometry.attributes.position.needsUpdate = true;
}

const SND = { ctx: null };
const midi = n => 440 * Math.pow(2, (n - 69) / 12), CHORDS = [[45, 52, 57, 60, 64], [41, 48, 53, 57, 60], [48, 55, 60, 64, 67], [43, 50, 55, 59, 62]];
function sndInit() {
  if (SND.ctx) { if (SND.ctx.state === 'suspended') SND.ctx.resume(); return; }
  try {
    const C = new (window.AudioContext || window.webkitAudioContext)(), out = C.createDynamicsCompressor(); out.connect(C.destination);
    const master = C.createGain(); master.gain.value = 0; const f = C.createBiquadFilter(); f.type = 'lowpass'; f.frequency.value = 300; f.connect(master); master.connect(out);
    const og = C.createGain(); og.gain.value = .5; og.connect(f); for (const fr of [48, 48.6, 72.3]) { const o = C.createOscillator(); o.type = 'sawtooth'; o.frequency.value = fr; o.connect(og); o.start(); }
    const len = C.sampleRate * 2, buf = C.createBuffer(1, len, C.sampleRate), d = buf.getChannelData(0); for (let i = 0; i < len; i++) d[i] = Math.random() * 2 - 1;
    const ns = C.createBufferSource(); ns.buffer = buf; ns.loop = true; const bp = C.createBiquadFilter(); bp.type = 'bandpass'; bp.frequency.value = 700; bp.Q.value = .6; const ng = C.createGain(); ng.gain.value = 0; ns.connect(bp); bp.connect(ng); ng.connect(out); ns.start();
    const mus = C.createGain(); mus.gain.value = 0; const mf = C.createBiquadFilter(); mf.type = 'lowpass'; mf.frequency.value = 1500; mus.connect(mf); mf.connect(out);
    Object.assign(SND, { ctx: C, out, master, f, ng, bp, mus, noise: buf, next: 0, ci: 0 });
  } catch (e) { SND.ctx = null; }
}
function sndUpdate() {
  if (!SND.ctx) return; const C = SND.ctx, t = C.currentTime, on = RIDE.on && RIDE.sound, s = Math.max(RIDE.speedN, RIDE.jumpN);
  SND.master.gain.setTargetAtTime(on ? .035 + .05 * s : 0, t, .2); SND.f.frequency.setTargetAtTime(240 + 900 * s, t, .2); SND.ng.gain.setTargetAtTime(on ? .04 * s + .1 * RIDE.jumpN : 0, t, .15); SND.bp.frequency.setTargetAtTime(500 + 1500 * s, t, .2);
  SND.mus.gain.setTargetAtTime(on ? 1 : 0, t, .8);
  if (on && t > SND.next - .3) { const t0 = Math.max(t, SND.next), ch = CHORDS[SND.ci++ % 4];       // a slow, cinematic chord loop
    for (const n of ch) { const o = C.createOscillator(), g = C.createGain(); o.type = n < 50 ? 'sawtooth' : 'triangle'; o.frequency.value = midi(n); o.detune.value = (Math.random() - .5) * 10;
      g.gain.setValueAtTime(0, t0); g.gain.linearRampToValueAtTime(n < 50 ? .02 : .011, t0 + 1.6); g.gain.linearRampToValueAtTime(0, t0 + 5.4); o.connect(g); g.connect(SND.mus); o.start(t0); o.stop(t0 + 5.6); }
    SND.next = t0 + 4.6; }
}
function sfx(kind) {
  if (!SND.ctx || !RIDE.sound || !RIDE.on) return; const C = SND.ctx, t = C.currentTime;
  const tone = (type, f0, f1, dur, vol) => { const o = C.createOscillator(), g = C.createGain(); o.type = type; o.frequency.setValueAtTime(f0, t); o.frequency.exponentialRampToValueAtTime(f1, t + dur); g.gain.setValueAtTime(vol, t); g.gain.exponentialRampToValueAtTime(.0001, t + dur); o.connect(g); g.connect(SND.out); o.start(t); o.stop(t + dur + .02); };
  const noise = (fc, dur, vol, type) => { const s = C.createBufferSource(); s.buffer = SND.noise; const fl = C.createBiquadFilter(); fl.type = type || 'lowpass'; fl.frequency.setValueAtTime(fc, t); fl.frequency.exponentialRampToValueAtTime(Math.max(60, fc * .15), t + dur); const g = C.createGain(); g.gain.setValueAtTime(vol, t); g.gain.exponentialRampToValueAtTime(.0001, t + dur); s.connect(fl); fl.connect(g); g.connect(SND.out); s.start(t, Math.random()); s.stop(t + dur + .05); };
  if (kind === 'pew') tone('square', 1500, 240, .16, .035); else if (kind === 'zap') tone('sawtooth', 700, 120, .22, .03);
  else if (kind === 'boom') { noise(1800, 1.1, .35); tone('sine', 90, 30, .8, .25); } else if (kind === 'hit') { noise(900, .35, .2); tone('sine', 160, 60, .3, .15); }
  else if (kind === 'jump') { noise(400, 1.6, .18, 'bandpass'); tone('sawtooth', 60, 520, 1.4, .05); } else if (kind === 'drop') { noise(2500, .9, .28); tone('sine', 120, 40, .7, .2); }
  else if (kind === 'alarm') { tone('square', 880, 860, .14, .03); setTimeout(() => RIDE.on && RIDE.sound && tone('square', 660, 650, .14, .03), 180); } else if (kind === 'blip') tone('sine', 880, 1320, .3, .04); else if (kind === 'comm') { noise(3000, .25, .08, 'highpass'); tone('square', 420, 380, .25, .025); }
}

function spaceBodies(skip) {
  const out = []; for (const o of planets) { if (!o.valid || o === skip) continue; out.push({ pos: o.pos, r: o.group.scale.x * (o.meta.type === 3 ? 2.5 : 1.7) + .35 }); } out.push({ pos: SUN_POS, r: SUN_R * 1.7 + .5 }); return out;
}
function bestHover(p, n) {
  const r = p.group.scale.x, out = p.pos.clone().setY(0).normalize(), side = new V3().crossVectors(AXY, out), base = n % 2 ? .85 : -.85, others = spaceBodies(p); let best = null, bs = 1e9;
  for (const dm of [3.1, 3.9, 5.0, 6.4]) for (const el of [.35, 1.3, 2.4, 3.8, 5.6]) for (let k = 0; k < 24; k++) {
    const d = r * dm + .9, th = k / 24 * 6.2832, pos = p.pos.clone().addScaledVector(out, Math.cos(th) * d).addScaledVector(side, Math.sin(th) * d).addScaledVector(AXY, r * el), v = p.pos.clone().sub(pos), dT = v.length(); v.multiplyScalar(1 / dT); const angT = Math.asin(clamp(r / dT, 0, 1)); let pen = 0;
    for (const o of others) { const vec = o.pos.clone().sub(pos), dist = vec.length(); if (dist < o.r + 1.6) pen += 80; const ang = Math.acos(clamp(vec.dot(v) / dist, -1, 1)), angO = Math.asin(clamp((o.r / 1.3) / dist, 0, 1)), over = angT * 1.3 + angO * 1.5 + .3 - ang;
      if (over > 0) pen += over * (1 + angO / angT) * 16; if (ang < 1.0 && angO > angT * .7) pen += (angO / angT) * (1.0 - ang) * 5; }
    pen += Math.abs(Math.atan2(Math.sin(th - base), Math.cos(th - base))) * .12 + (dm - 3.1) * .18 + el * .06;
    if (pen < bs) { bs = pen; best = pos; } }
  bestHover.last = bs; return best;
}
function avoidBodies(P0, P1, P2, P3, skip) {
  const bodies = spaceBodies(skip), t3 = new V3(), dv = new V3();
  for (let it = 0; it < 12; it++) { let hit = false;
    for (let s = 1; s < 48; s++) { const t = s / 48; bez(P0, P1, P2, P3, t, t3);
      for (const b of bodies) { dv.copy(t3).sub(b.pos); const dist = dv.length(); if (dist >= b.r) continue; hit = true; if (dist < 1e-3) dv.set(0, 1, 0); else dv.multiplyScalar(1 / dist); const B1 = 3 * (1 - t) * (1 - t) * t, B2 = 3 * (1 - t) * t * t, sum = Math.max(.18, B1 + B2), push = (b.r - dist) * 1.3 / sum;
        P1.addScaledVector(dv, push); P2.addScaledVector(dv, push); } }
    if (!hit) break; }
}

function hoverFor(st, n) {
  const o = { pos: new V3(), face: new V3() };
  if (st.type === 'intro') { const th = CAM.th + .5, ph = .5, d = 21; o.pos.set(Math.cos(ph) * Math.sin(th) * d, Math.sin(ph) * d, Math.cos(ph) * Math.cos(th) * d); }
  else if (st.type === 'sun') { let best = null, bs = 1e9; const bodies = spaceBodies(null).filter(b => b.pos !== SUN_POS); for (const d of [3.3, 4.2, 5.2]) for (let k = 0; k < 24; k++) { const th = CAM.th + 1.7 + k / 24 * 6.2832, pos = new V3(Math.sin(th) * d, .55, Math.cos(th) * d); let pen = (d - 3.3) * .4 + Math.abs(Math.atan2(Math.sin(th - CAM.th - 1.7), Math.cos(th - CAM.th - 1.7))) * .1; for (const b of bodies) { const dist = pos.distanceTo(b.pos); if (dist < b.r + 1.5) pen += 100; else pen += 6 / (dist - b.r); } if (pen < bs) { bs = pen; best = pos; } } o.pos.copy(best); }
  else if (st.type === 'zone') { const F = FIELD * .55, c = new V3(ZC[st.q][0] * F, 0, ZC[st.q][1] * F); o.pos.copy(c).multiplyScalar(1.35).add(new V3(0, 3.4, 0)); o.face.copy(c).multiplyScalar(.55); }

  else { o.pos.copy(SHIP.pos).addScaledVector(SHIP.fwd, 9).add(new V3(0, 3, 0)); o.face.copy(o.pos).addScaledVector(SHIP.fwd, 10).add(new V3(0, 2, 0)); }

  return o;
}

/* ---- flight model: constant-speed cruise along an arc-length-parametrized curve, with real acceleration and braking ---- */
const bez = (a, b, c, d, t, out) => { const u = 1 - t; return out.set(0, 0, 0).addScaledVector(a, u * u * u).addScaledVector(b, 3 * u * u * t).addScaledVector(c, 3 * u * t * t).addScaledVector(d, t * t * t); };
function vprof(x, jump, dive, cruise) {        // relative speed over normalized leg time
  if (dive) return .1 + .9 * Math.pow(x, 1.6);
  if (cruise === 1) return 1; if (cruise === 2) return Math.min(1, .35 + x * 5); if (cruise === 3) return x < .6 ? 1 : .02 + .98 * ease5((1 - x) / .4);
  if (jump) { if (x < .24) return .04 + .2 * ease5(x / .24); if (x < .8) return .24 + 6 * sstep(.24, .32, x); return (6 * (1 - sstep(.8, .87, x)) + .24) * (1 - sstep(.87, 1, x)) + .015; }
  const a = .24, d = .3; return x < a ? .06 + .94 * ease5(x / a) : x > 1 - d ? .015 + .985 * ease5((1 - x) / d) : 1;
}
function planLeg(st, i) {
  const h = hoverFor(st, i), P0 = SHIP.base.clone(), P3 = h.pos, D = P0.distanceTo(P3), toDest = P3.clone().sub(P0).normalize(), arrive = h.face.clone().sub(P3).normalize();
  const jump = st.type === 'intro' || (D > 16 && !st.threat && st.type !== 'outro'), k = D * .34;
  const P1 = P0.clone().addScaledVector(_a.copy(SHIP.fwd).multiplyScalar(.55).addScaledVector(toDest, .45).addScaledVector(AXY, .22).normalize(), k);
  const P2 = P3.clone().addScaledVector(arrive, -k).addScaledVector(AXY, Math.min(3, D * .08));
  let dur = jump ? 3.6 : clamp(2.8 + D * .075, 3.2, 6.5); if (st.threat) dur = Math.max(dur, 6.5);
  avoidBodies(P0, P1, P2, P3, st.type === 'planet' ? st.p : null);
  const lg = makeLeg(P0, P1, P2, P3, h.face, arrive, jump, dur, false);
  // shot plan: some planet arrivals cut to a fly-by camera waiting near the end of the path

  lg.flyby = flyby; return lg;
}
function makeLeg(P0, P1, P2, P3, face, arrive, jump, dur, dive, cruise) {
  const N = 120, ss = new Float32Array(N + 1), pv = P0.clone(), tp = new V3(); let acc = 0;
  for (let j = 1; j <= N; j++) { bez(P0, P1, P2, P3, j / N, tp); acc += tp.distanceTo(pv); pv.copy(tp); ss[j] = acc; }
  const M = 240, sd = new Float32Array(M + 1); let s = 0; for (let j = 1; j <= M; j++) { s += vprof((j - .5) / M, jump, dive, cruise) / M; sd[j] = s; } for (let j = 0; j <= M; j++) sd[j] = sd[j] / s * acc;
  return { P0, P1, P2, P3, face, arrive, jump, dur, len: acc, ss, sd, N, M, flyby: false, dive: !!dive, cruise: cruise || 0, fb: null };
}
const _lt = new V3(), _lt2 = new V3(), _lt3 = new V3();
function legSample(leg, u, out) {           // u = leg time fraction -> position on the curve, curve parameter t, speed
  const fi = clamp(u, 0, 1) * leg.M, i = Math.min(leg.M - 1, Math.floor(fi)), f = fi - i, s = leg.sd[i] + (leg.sd[i + 1] - leg.sd[i]) * f;
  let lo = 0, hi = leg.N; while (lo < hi) { const m = (lo + hi) >> 1; if (leg.ss[m] < s) lo = m + 1; else hi = m; }
  const k = Math.max(1, lo), s0 = leg.ss[k - 1], s1 = leg.ss[k], t = (k - 1 + (s1 > s0 ? (s - s0) / (s1 - s0) : 0)) / leg.N;
  bez(leg.P0, leg.P1, leg.P2, leg.P3, t, out); return t;
}
function curveFrame(leg, t, tan, curv) {     // unit tangent and curvature vector of the Bezier at t
  const e = .004, a = bez(leg.P0, leg.P1, leg.P2, leg.P3, Math.max(0, t - e), _lt), b = bez(leg.P0, leg.P1, leg.P2, leg.P3, Math.min(1, t + e), _lt2), c = bez(leg.P0, leg.P1, leg.P2, leg.P3, t, _lt3);
  tan.copy(b).sub(a); const ds = tan.length() / 2; tan.normalize();
  curv.copy(a).add(b).addScaledVector(c, -2).multiplyScalar(1 / Math.max(ds * ds, 1e-6)); curv.addScaledVector(tan, -curv.dot(tan));
}

function surfPack(items) {                       // greedy circle packing on a golden-angle spiral: biggest company at the centre
  const placed = []; for (let i = 0; i < items.length; i++) { const it = items[i], rr = it.R * 1.22;
    if (!i) { it.x = 0; it.z = 0; placed.push(it); continue; }
    for (let k = 1; k < 4000; k++) { const a = k * 2.399963, rad = .32 * Math.sqrt(k) * 1.0, x = Math.cos(a) * rad, z = Math.sin(a) * rad; let ok = true; for (const o of placed) { const dx = x - o.x, dz = z - o.z, md = rr + o.R * 1.22 + .5; if (dx * dx + dz * dz < md * md) { ok = false; break; } } if (ok) { it.x = x; it.z = z; placed.push(it); break; } }
    if (it.x === undefined) { it.x = 0; it.z = 0; } }
}
