function rolling(a, n) {
  const mean = new Float64Array(a.length).fill(NaN), sd = new Float64Array(a.length).fill(NaN); let s = 0, s2 = 0, c = 0;
  for (let i = 0; i < a.length; i++) { const v = a[i]; if (v === v) { s += v; s2 += v * v; c++; } if (i >= n) { const u = a[i - n]; if (u === u) { s -= u; s2 -= u * u; c--; } } if (i >= n - 1 && c === n) { const m = s / n; mean[i] = m; sd[i] = Math.sqrt(Math.max(0, s2 / n - m * m)); } }
  return { mean, sd };
}
function computeRRG() {

  for (const s of SEC) {

    for (let w = 0; w < W; w++) price[w] = a[D.wk[w]];

    const d = new Float64Array(W).fill(NaN); for (let w = K; w < W; w++) d[w] = ratio[w] - ratio[w - K];
    const ds = rolling(d, WIN), mom = new Float64Array(W).fill(NaN); for (let w = 0; w < W; w++) if (ds.sd[w] > 0) mom[w] = 100 + d[w] / ds.sd[w];
    let v0 = 0; while (v0 < W && !(ratio[v0] === ratio[v0] && mom[v0] === mom[v0])) v0++;
    R.s[s.t] = { price, rs, ratio, mom, v0 };

}

  vec4 j=p-49.0*floor(p*ns.z*ns.z); vec4 x_=floor(j*ns.z); vec4 y_=floor(j-7.0*x_);
  vec4 x=x_*ns.x+ns.yyyy; vec4 y=y_*ns.x+ns.yyyy; vec4 h=1.0-abs(x)-abs(y);
  vec4 b0=vec4(x.xy,y.xy); vec4 b1=vec4(x.zw,y.zw);

function frame(now) {

  const dtRaw = Math.min(.05, raw / 1000); dtS += (dtRaw - dtS) * .2; const dt = dtS; lastT = now; const tm = now / 1000; U.time.value = tm;
  if (T.playing) { T.target = Math.min(T.hi, T.target + dt * SPEEDS[T.speed][0]); if (T.target >= T.hi) { if (T.loop) { T.target = T.cur = T.lo; } else { T.playing = false; $('#playIcon').innerHTML = '<path d="M3 1.5v13l11-6.5z"/>'; } } }

      else PERF.good = 0; } }
}

function spawnRocket(r) {
  const ang = Math.random() * 6.2832, d = new V3(Math.cos(ang), (Math.random() - .5) * .3, Math.sin(ang)).normalize(), o = new V3().crossVectors(d, AXY).normalize().multiplyScalar((Math.random() < .5 ? -1 : 1) * (16 + Math.random() * 12)); o.y += (Math.random() - .5) * 14;
  r.S = o.clone().addScaledVector(d, -62); r.E = o.clone().addScaledVector(d, 62); r.dir = d; r.dur = 16 + Math.random() * 10; r.t = -(2 + Math.random() * 5); r.scale = .62 + Math.random() * .28;
}

function cometAt(t, out) {
  const a = 26, e = .54, M = (t / 200) * 6.2832; let E = M; for (let i = 0; i < 5; i++) E = E - (E - e * Math.sin(E) - M) / (1 - e * Math.cos(E));
  out.set(a * (Math.cos(E) - e), 0, a * Math.sqrt(1 - e * e) * Math.sin(E)).applyAxisAngle(AXY, .4).applyAxisAngle(AXX, .55).applyAxisAngle(AXY, 1.1); return out;
}

  const q = FXQ[QUAL.lvl];
  for (const m of moons) { if (!m.mesh.visible) continue; const a = m.ph + tm * m.sp; m.mesh.position.set(Math.cos(a) * m.orb, Math.sin(a) * m.orb * Math.sin(m.inc), Math.sin(a) * m.orb * Math.cos(m.inc)); m.mesh.rotation.y = tm * .1; m.uni.uPxR.value = (m.p.uni.uPxR.value || 20) * m.sz; }

  for (const r of rockets) {

  }

    const head = _a.copy(mt.p0).addScaledVector(mt.dir, mt.dist * mt.t), tail = _b.copy(head).addScaledVector(mt.dir, -70 * Math.min(1, mt.t * 4)); _d.copy(camera.position).sub(head); _c.crossVectors(mt.dir, _d).normalize().multiplyScalar(.8);

// A deliberately plain re-computation: direct loops, two-pass mean and standard deviation, no rolling tricks.
function naive(t, w) {
  const o = R.s[t], price = o.price, spy = R.spy, { WIN, SM, K } = PARAM;
  const rs = j => price[j] / spy[j];
  const rsm = j => { let s = 0; for (let i = 0; i < SM; i++) s += rs(j - i); return s / SM; };
  const stats = arr => { const n = arr.length, m = arr.reduce((a, b) => a + b, 0) / n; let v = 0; for (const x of arr) v += (x - m) * (x - m); return { m, sd: Math.sqrt(v / n) }; };
  const ratioAt = j => { const vals = []; for (let k = j - WIN + 1; k <= j; k++) vals.push(rsm(k)); const { m, sd } = stats(vals); return { ratio: 100 + (rsm(j) - m) / sd, rsm: rsm(j), mean: m, sd }; };
  const cur = ratioAt(w), ds = []; for (let k = w - WIN + 1; k <= w; k++) ds.push(ratioAt(k).ratio - ratioAt(k - K).ratio);
  const dst = stats(ds), dNow = ds[ds.length - 1];
  return { rs: [rs(w - 2), rs(w - 1), rs(w)], rsm: cur.rsm, mean: cur.mean, sd: cur.sd, ratio: cur.ratio, ratioK: ratioAt(w - K).ratio, d: dNow, dsd: dst.sd, mom: 100 + dNow / dst.sd, price: price[w], spy: spy[w] };
}
function runSelfCheck() {
  let n = 0, maxR = 0, maxM = 0, zoneFlips = 0;
  for (const s of SEC) { const o = R.s[s.t]; if (!o) continue; for (let w = o.v0; w < R.W; w++) { const q = naive(s.t, w), dr = Math.abs(q.ratio - o.ratio[w]), dm = Math.abs(q.mom - o.mom[w]); n++; if (dr > maxR) maxR = dr; if (dm > maxM) maxM = dm; if (quad(q.ratio, q.mom) !== quad(o.ratio[w], o.mom[w])) zoneFlips++; } }
  return { n, maxR, maxM, zoneFlips };
}
function auditOne(t, wk, daily) {
  let first = -1, missing = 0, valid = 0, maxMove = 0, maxW = -1;
  for (let w = 0; w < wk.length; w++) { const v = wk[w]; if (v === v) { if (first < 0) first = w; valid++; if (w > 0 && wk[w - 1] === wk[w - 1]) { const m = wk[w] / wk[w - 1] - 1; if (Math.abs(m) > Math.abs(maxMove)) { maxMove = m; maxW = w; } } } else if (first >= 0) missing++; }
  let df = -1, dl = -1; for (let i = 0; i < daily.length; i++) if (daily[i] === daily[i]) { if (df < 0) df = i; dl = i; }
  const o = R.s[t];
  return { t, first: D.dates[df], last: D.dates[dl], valid, missing, rrg: o ? R.dates[o.v0] : '(benchmark)', maxMove, maxDate: maxW >= 0 ? R.dates[maxW] : '--' };
}

  <h3 id="m-measure">What it measures (and what it doesn't)</h3>
  <p>Relative strength is not how much a sector rose. It asks a different question: <b>did the sector do better or worse than the market?</b> If the market gains 2% and Utilities gain 3%, Utilities gained relative strength. If the market falls 5% and Staples fall 2%, Staples gained relative strength even though they lost money. That is why a sector can be "Leading" while its price is falling.</p>
  <p>The chart shows <b>rotation</b>, meaning where money is flowing between sectors. It does not predict prices, and it says nothing about whether the whole market will rise or fall.</p>

  <h3 id="m-math">The math, step by step</h3>
  <p>For each sector and each week <i>w</i>, with <i>P</i> the sector's weekly close and <i>B</i> SPY's:</p>
  <div class="formula"><b>1. Relative strength</b>&nbsp; RS(w) = P(w) ÷ B(w)<br><b>2. Smooth</b>&nbsp; RSm(w) = average of RS over the last ${SM} weeks<br><b>3.   <p><b>Why normalize?</b> Dividing by the standard deviation turns each sector's relative strength into "how unusual is this versus its own recent range?" A reading of 102 means about two standard deviations above its half-year norm. <b>Why does it rotate clockwise?</b> Momentum is the speed of the ratio, so it turns before the ratio does. That is what makes sectors loop through the zones in order.</p>
